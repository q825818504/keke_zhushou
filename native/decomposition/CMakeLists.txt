cmake_minimum_required(VERSION 3.24)
project(keke_native_decomposition LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "This researched ABI requires Windows x64")
endif()
add_library(keke_native_decomposition STATIC decomposition.cpp)
target_compile_features(keke_native_decomposition PUBLIC cxx_std_17)
target_include_directories(keke_native_decomposition PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(keke_native_decomposition PRIVATE bcrypt)
if(MSVC)
    target_compile_options(keke_native_decomposition PRIVATE /W4 /permissive- /EHsc)
endif()
add_library(keke_native_decomposition_dll SHARED decomposition.cpp)
target_compile_features(keke_native_decomposition_dll PUBLIC cxx_std_17)
target_compile_definitions(keke_native_decomposition_dll PRIVATE KEKE_DECOMPOSITION_DLL)
target_link_libraries(keke_native_decomposition_dll PRIVATE bcrypt)
set_target_properties(keke_native_decomposition_dll PROPERTIES OUTPUT_NAME KekeNativeDecomposition)
